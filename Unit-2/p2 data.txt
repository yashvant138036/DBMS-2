-- Create the main Employee table
CREATE TABLE emp (
    empno NUMBER PRIMARY KEY,
    ename VARCHAR2(50),
    sal NUMBER,
    deptno NUMBER
);

-- Create the Audit/Update table
CREATE TABLE emp_update (
    empno NUMBER,
    raise_amount NUMBER,
    update_date DATE
);
